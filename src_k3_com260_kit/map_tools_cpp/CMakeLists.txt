cmake_minimum_required(VERSION 3.8)
project(map_tools_cpp)
find_package(ament_cmake REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc)
find_package(yaml-cpp REQUIRED)
foreach(node map_postprocessor map_evaluator map_coverage)
  add_executable(${node} src/${node}.cpp)
  target_compile_features(${node} PRIVATE cxx_std_17)
  target_include_directories(${node} PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(${node} ${OpenCV_LIBS} yaml-cpp)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
