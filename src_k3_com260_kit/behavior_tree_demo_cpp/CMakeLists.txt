cmake_minimum_required(VERSION 3.8)
project(behavior_tree_demo_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(recovery_demo_cpp REQUIRED)
find_package(behaviortree_cpp_v3 REQUIRED)
foreach(node bt_client custom_behavior_tree)
  add_executable(${node} src/${node}.cpp)
  target_compile_features(${node} PUBLIC cxx_std_17)
  target_compile_options(${node} PRIVATE -UNDEBUG)
  target_include_directories(${node} PRIVATE include)
  ament_target_dependencies(${node} rclcpp rclcpp_action nav2_msgs std_msgs recovery_demo_cpp behaviortree_cpp_v3)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
