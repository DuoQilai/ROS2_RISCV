cmake_minimum_required(VERSION 3.8)
project(hybrid_astar_planner)
find_package(ament_cmake REQUIRED)
find_package(grid_path_planner REQUIRED)
add_executable(${PROJECT_NAME} src/${PROJECT_NAME}.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PRIVATE include)
ament_target_dependencies(${PROJECT_NAME} grid_path_planner)
install(TARGETS ${PROJECT_NAME} DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(grid_path_planner)
if(BUILD_TESTING)
  enable_testing()
  add_executable(known_inputs test/known_inputs.cpp)
  target_compile_features(known_inputs PUBLIC cxx_std_17)
  target_compile_options(known_inputs PRIVATE -UNDEBUG)
  target_include_directories(known_inputs PRIVATE include)
  ament_target_dependencies(known_inputs grid_path_planner)
  add_test(NAME known_inputs COMMAND known_inputs)
endif()
ament_package()
