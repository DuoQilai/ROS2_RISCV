cmake_minimum_required(VERSION 3.8)
project(navigation_client_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_msgs REQUIRED)
foreach(node bt_navigator_client multi_goal_navigation)
  add_executable(${node} src/${node}.cpp)
  target_compile_features(${node} PUBLIC cxx_std_17)
  ament_target_dependencies(${node} rclcpp rclcpp_action nav2_msgs)
  target_include_directories(${node} PRIVATE include)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY behavior_trees DESTINATION share/${PROJECT_NAME})
ament_package()
