cmake_minimum_required(VERSION 3.8)
project(multi_robot_scheduler)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(recovery_demo_cpp REQUIRED)
find_package(yaml-cpp REQUIRED)
foreach(node multi_robot_scheduler)
  add_executable(${node} src/${node}.cpp)
  target_compile_features(${node} PUBLIC cxx_std_17)
  target_compile_options(${node} PRIVATE -UNDEBUG)
  target_include_directories(${node} PRIVATE include)
  ament_target_dependencies(${node} rclcpp rclcpp_action nav2_msgs recovery_demo_cpp)
  target_link_libraries(${node} yaml-cpp)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
