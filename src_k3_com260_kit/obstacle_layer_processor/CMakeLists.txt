cmake_minimum_required(VERSION 3.8)
project(obstacle_layer_processor)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(costmap_demo REQUIRED)
add_executable(obstacle_layer_processor src/obstacle_layer_processor.cpp)
target_compile_features(obstacle_layer_processor PUBLIC cxx_std_17)
ament_target_dependencies(obstacle_layer_processor rclcpp sensor_msgs geometry_msgs tf2_ros tf2_geometry_msgs costmap_demo)
install(TARGETS obstacle_layer_processor DESTINATION lib/${PROJECT_NAME})
ament_package()
