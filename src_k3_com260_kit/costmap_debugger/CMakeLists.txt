cmake_minimum_required(VERSION 3.8)
project(costmap_debugger)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
add_executable(costmap_debugger src/costmap_debugger.cpp)
target_compile_features(costmap_debugger PUBLIC cxx_std_17)
ament_target_dependencies(costmap_debugger rclcpp nav_msgs)
install(TARGETS costmap_debugger DESTINATION lib/${PROJECT_NAME})
ament_package()
