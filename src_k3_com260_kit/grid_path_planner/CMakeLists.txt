cmake_minimum_required(VERSION 3.8)
project(grid_path_planner)
find_package(ament_cmake REQUIRED)
find_package(costmap_demo REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
add_executable(${PROJECT_NAME} src/${PROJECT_NAME}.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PRIVATE include)
ament_target_dependencies(${PROJECT_NAME} costmap_demo)
install(TARGETS ${PROJECT_NAME} DESTINATION lib/${PROJECT_NAME})
add_executable(grid_planner_node src/grid_planner_node.cpp)
target_compile_features(grid_planner_node PUBLIC cxx_std_17)
target_include_directories(grid_planner_node PRIVATE include)
ament_target_dependencies(grid_planner_node costmap_demo rclcpp nav_msgs)
install(TARGETS grid_planner_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(costmap_demo)
if(BUILD_TESTING)
  enable_testing()
  add_executable(known_inputs test/known_inputs.cpp)
  target_compile_features(known_inputs PUBLIC cxx_std_17)
  target_compile_options(known_inputs PRIVATE -UNDEBUG)
  target_include_directories(known_inputs PRIVATE include)
  ament_target_dependencies(known_inputs costmap_demo)
  add_test(NAME known_inputs COMMAND known_inputs)
endif()
ament_package()
