cmake_minimum_required(VERSION 3.8)
project(camera_geometry)
find_package(ament_cmake REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core calib3d)
foreach(node camera_geometry)
  add_executable(${node} src/${node}.cpp)
  target_compile_features(${node} PUBLIC cxx_std_17)
  target_compile_options(${node} PRIVATE -UNDEBUG)
  target_include_directories(${node} PRIVATE include)
  target_link_libraries(${node} ${OpenCV_LIBS})
  target_include_directories(${node} PRIVATE ${OpenCV_INCLUDE_DIRS})
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
if(BUILD_TESTING)
  enable_testing()
  add_test(NAME known_inputs COMMAND camera_geometry)
endif()
ament_package()
