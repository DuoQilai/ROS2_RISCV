cmake_minimum_required(VERSION 3.8)
project(global_planner_client)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
add_executable(global_planner_client src/global_planner_client.cpp)
target_compile_features(global_planner_client PUBLIC cxx_std_17)
ament_target_dependencies(global_planner_client rclcpp rclcpp_action nav2_msgs nav_msgs)
install(TARGETS global_planner_client DESTINATION lib/${PROJECT_NAME})
ament_package()
