cmake_minimum_required(VERSION 3.8)
project(visual_slam_eval)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
find_path(ORB_SLAM3_ROOT include/System.h HINTS ENV ORB_SLAM3_ROOT REQUIRED)
add_executable(bag_vio src/bag_vio.cpp)
target_compile_features(bag_vio PRIVATE cxx_std_17)
target_include_directories(bag_vio PRIVATE ${ORB_SLAM3_ROOT}/include
  ${ORB_SLAM3_ROOT}/include/CameraModels ${ORB_SLAM3_ROOT}/Thirdparty/Sophus ${ORB_SLAM3_ROOT})
ament_target_dependencies(bag_vio rclcpp rosbag2_cpp sensor_msgs nav_msgs geometry_msgs tf2_ros cv_bridge)
target_link_libraries(bag_vio Eigen3::Eigen ${ORB_SLAM3_ROOT}/lib/libORB_SLAM3.so ${Pangolin_LIBRARIES})
install(TARGETS bag_vio DESTINATION lib/${PROJECT_NAME})
ament_package()
