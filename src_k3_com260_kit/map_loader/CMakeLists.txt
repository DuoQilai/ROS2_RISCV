cmake_minimum_required(VERSION 3.8)
project(map_loader)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(map_loader src/map_loader.cpp)
target_compile_features(map_loader PUBLIC cxx_std_17)
ament_target_dependencies(map_loader rclcpp nav_msgs nav2_msgs)
target_link_libraries(map_loader yaml-cpp)
install(TARGETS map_loader DESTINATION lib/${PROJECT_NAME})
ament_package()
