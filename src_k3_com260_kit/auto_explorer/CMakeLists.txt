cmake_minimum_required(VERSION 3.8)
project(auto_explorer)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(OpenCV REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(auto_explorer src/auto_explorer.cpp)
target_compile_features(auto_explorer PRIVATE cxx_std_17)
target_include_directories(auto_explorer PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(auto_explorer ament_index_cpp rclcpp rclcpp_action geometry_msgs sensor_msgs nav_msgs nav2_msgs tf2_ros)
target_link_libraries(auto_explorer ${OpenCV_LIBS} yaml-cpp)
install(TARGETS auto_explorer DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(frontiers_test test/frontiers_test.cpp)
  target_compile_features(frontiers_test PRIVATE cxx_std_17)
  target_include_directories(frontiers_test PRIVATE include ${OpenCV_INCLUDE_DIRS})
  ament_target_dependencies(frontiers_test nav_msgs)
  target_link_libraries(frontiers_test ${OpenCV_LIBS})
  add_test(NAME frontiers_test COMMAND frontiers_test)
endif()
ament_package()
