cmake_minimum_required(VERSION 3.8)
project(orbslam3)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(image_transport REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
find_path(ORB_SLAM3_ROOT include/System.h HINTS ENV ORB_SLAM3_ROOT REQUIRED)
add_executable(rgbd src/rgbd.cpp)
target_compile_features(rgbd PRIVATE cxx_std_17)
target_include_directories(rgbd PRIVATE ${ORB_SLAM3_ROOT}/include
  ${ORB_SLAM3_ROOT}/include/CameraModels ${ORB_SLAM3_ROOT}/Thirdparty/Sophus ${ORB_SLAM3_ROOT})
ament_target_dependencies(rgbd rclcpp sensor_msgs geometry_msgs nav_msgs cv_bridge message_filters image_transport tf2_ros)
target_link_libraries(rgbd Eigen3::Eigen ${ORB_SLAM3_ROOT}/lib/libORB_SLAM3.so ${Pangolin_LIBRARIES})
install(TARGETS rgbd DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
