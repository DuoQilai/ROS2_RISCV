cmake_minimum_required(VERSION 3.8)
project(sensor_fusion_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(image_transport REQUIRED)
find_package(Eigen3 REQUIRED)
foreach(node sensor_sync imu_odom_fusion lidar_pose_to_odom)
  add_executable(${node} src/${node}.cpp)
  target_compile_features(${node} PRIVATE cxx_std_17)
  ament_target_dependencies(${node} rclcpp sensor_msgs nav_msgs geometry_msgs message_filters image_transport)
  target_link_libraries(${node} Eigen3::Eigen)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
add_executable(fusion_math_demo src/fusion_math_demo.cpp)
target_compile_features(fusion_math_demo PRIVATE cxx_std_17)
target_include_directories(fusion_math_demo PRIVATE include)
target_link_libraries(fusion_math_demo Eigen3::Eigen)
install(TARGETS fusion_math_demo DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_test(NAME fusion_math_demo COMMAND fusion_math_demo)
endif()
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
if(EXISTS "/opt/ros/humble/share/cartographer/configuration_files")
  install(DIRECTORY /opt/ros/humble/share/cartographer/configuration_files/
    DESTINATION share/${PROJECT_NAME}/config FILES_MATCHING PATTERN "*.lua")
endif()
ament_package()
