cmake_minimum_required(VERSION 3.8)
project(fleet_scheduler)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(fleet_scheduler src/fleet_scheduler.cpp)
target_compile_features(fleet_scheduler PRIVATE cxx_std_17)
ament_target_dependencies(fleet_scheduler rclcpp rclcpp_action nav2_msgs nav_msgs geometry_msgs)
target_link_libraries(fleet_scheduler yaml-cpp)
install(TARGETS fleet_scheduler DESTINATION lib/${PROJECT_NAME})
add_executable(conflict_demo src/conflict_demo.cpp)
target_compile_features(conflict_demo PRIVATE cxx_std_17)
target_include_directories(conflict_demo PRIVATE include)
install(TARGETS conflict_demo DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_test(NAME conflict_demo COMMAND conflict_demo)
endif()
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
