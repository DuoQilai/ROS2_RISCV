cmake_minimum_required(VERSION 3.8)
project(orb_feature_extractor)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core features2d imgproc)
foreach(node orb_feature_extractor)
  add_executable(${node} src/${node}.cpp)
  target_compile_features(${node} PUBLIC cxx_std_17)
  target_compile_options(${node} PRIVATE -UNDEBUG)
  target_include_directories(${node} PRIVATE include)
  ament_target_dependencies(${node} rclcpp sensor_msgs cv_bridge)
  target_link_libraries(${node} ${OpenCV_LIBS})
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(known_inputs test/known_inputs.cpp)
  target_compile_features(known_inputs PUBLIC cxx_std_17)
  target_compile_options(known_inputs PRIVATE -UNDEBUG)
  target_include_directories(known_inputs PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(known_inputs ${OpenCV_LIBS})
  add_test(NAME known_inputs COMMAND known_inputs)
endif()
ament_package()
