cmake_minimum_required(VERSION 3.8)
project(custom_nav_component)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(custom_nav_component src/custom_nav_component.cpp)
target_compile_features(custom_nav_component PUBLIC cxx_std_17)
ament_target_dependencies(custom_nav_component rclcpp rclcpp_lifecycle geometry_msgs sensor_msgs)
install(TARGETS custom_nav_component DESTINATION lib/${PROJECT_NAME})
ament_package()
